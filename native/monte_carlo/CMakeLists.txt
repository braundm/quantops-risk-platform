cmake_minimum_required(VERSION 3.20)
project(quantops_monte_carlo LANGUAGES CXX)
include(CTest)
add_library(quantops_mc SHARED monte_carlo.cpp)
target_compile_features(quantops_mc PUBLIC cxx_std_17)
target_include_directories(quantops_mc PUBLIC .)
if(MSVC)
  target_compile_options(quantops_mc PRIVATE /W4 /WX /fp:precise)
else()
  target_compile_options(quantops_mc PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
if(BUILD_TESTING)
  add_executable(monte_carlo_tests tests.cpp)
  target_link_libraries(monte_carlo_tests PRIVATE quantops_mc)
  add_test(NAME monte_carlo COMMAND monte_carlo_tests)
endif()
